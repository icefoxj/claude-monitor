# Recordings to play instead of the synthesised alerts, in this build only:
# firmware/tab5/sounds/<theme>-<attention|failure>.wav (gitignored: nothing
# there is committed, and CI and the releases build without them). See
# firmware/tab5/sounds/README.md.
set(sound_dir "${CMAKE_CURRENT_LIST_DIR}/../sounds")
set(sound_files "")
set(sound_names "")
foreach(theme classic anime lcars galactic)
    foreach(alert attention failure)
        if(EXISTS "${sound_dir}/${theme}-${alert}.wav")
            list(APPEND sound_files "${sound_dir}/${theme}-${alert}.wav")
            list(APPEND sound_names "${theme}-${alert}")
        endif()
    endforeach()
endforeach()

idf_component_register(SRCS "main.cpp"
                    INCLUDE_DIRS "."
                    EMBED_FILES ${sound_files})

# monitor_sounds.h: the embedded files' symbols and which theme and alert
# each one stands for
set(decls "")
set(rows "")
foreach(name ${sound_names})
    string(REPLACE "-" ";" parts "${name}")
    list(GET parts 0 theme)
    list(GET parts 1 alert)
    string(SUBSTRING "${theme}" 0 1 head)
    string(SUBSTRING "${theme}" 1 -1 tail)
    string(TOUPPER "${head}" head)
    set(Theme "${head}${tail}")
    string(SUBSTRING "${alert}" 0 1 head)
    string(SUBSTRING "${alert}" 1 -1 tail)
    string(TOUPPER "${head}" head)
    set(Alert "${head}${tail}")
    string(MAKE_C_IDENTIFIER "${name}_wav" sym)
    string(APPEND decls "extern const uint8_t ${sym}_start[] asm(\"_binary_${sym}_start\");\n")
    string(APPEND decls "extern const uint8_t ${sym}_end[]   asm(\"_binary_${sym}_end\");\n")
    string(APPEND rows "    {monitor::Theme::${Theme}, monitor::Alert::${Alert}, ${sym}_start, ${sym}_end, \"${name}\"},\n")
endforeach()
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/monitor_sounds.h" CONTENT
"// Generated by firmware/tab5/main/CMakeLists.txt from the WAV files found
// in firmware/tab5/sounds/ when the build was configured
#pragma once
#include <cstdint>
#include \"monitor/sounds.h\"

${decls}
struct EmbeddedSound {
    monitor::Theme theme;
    monitor::Alert alert;
    const uint8_t* start;
    const uint8_t* end;
    const char*    name;   // nullptr ends the list
};

static const EmbeddedSound kEmbeddedSounds[] = {
${rows}    {monitor::Theme::Classic, monitor::Alert::Attention, nullptr, nullptr, nullptr},
};
" @ONLY)
target_include_directories(${COMPONENT_LIB} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")

# A file added to or removed from the folder configures the build again
if(EXISTS "${sound_dir}")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${sound_dir}")
endif()
